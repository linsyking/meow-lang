%%  theorem: on inputs whose letters are pairwise distinct and avoid the

%% \section{Separable inputs: total rigidity at every depth}
%% \label{sec:separable}
%
% Conventions: as in the prov machinery of rounds 10-11 -- values are
% strings of atoms $(c,\ell)$ with $\ell \in \{0,\dots,n{-}1\}$ an input
% offset or $\ell = \bot$ for a constant character; $\mathrm{prov}$ is
% the label sequence of the output; $\Gamma(E)$ is the finite set of
% letters occurring in $E$'s constants; $S$-depth counts pass nodes.
%
% \begin{definition}[Separable]
% \label{def:separable}
% An input $w = x_0 x_1 \cdots x_{n-1}$ is \emph{separable for $E$} if
% its letters are pairwise distinct and disjoint from $\Gamma(E)$.
% \end{definition}
%
% \begin{definition}[Copy form]
% \label{def:copyform}
% A value is in \emph{copy form} (with $M \geq 0$ copies) if it is
% $v_0\,W\,v_1\,W\,v_2 \cdots W\,v_M$ where $W$ denotes a contiguous
% block of $n$ atoms spelling $w$ with labels $(0,1,\dots,n{-}1)$ in
% order, and each $v_j \in \Gamma(E)^*$ is unlabeled.  The blocks $W$
% are the \emph{instances}.  For $M = 0$ the value is a constant word.
% \end{definition}
%
% The key structural fact is that on separable inputs this form is
% closed under the pass.
%
% \begin{lemma}[Copy-alignment]
% \label{lem:copyalign}
% Let $w$ be separable, let $T = v_0 W v_1 \cdots W v_M$ and
% $X = u_0 W u_1 \cdots W u_{\pi}$ be in copy form ($\pi \geq 0$,
% $X \neq \epsilon$).  Then every occurrence of $X$ in $T$ either
% ($\pi = 0$) lies inside a single constant run $v_j$, or ($\pi \geq 1$)
% covers $\pi$ \emph{consecutive instances entirely} -- instance $a$
% through $a + \pi - 1$ -- together with the $|u_0|$-letter suffix of
% $v_{a-1}$ and the $|u_\pi|$-letter prefix of $v_{a+\pi-1}$, the
% interior constant runs matching $X$'s gaps exactly.  No match window
% starts or ends strictly inside an instance.
% \end{lemma}
%
% \begin{proof}
% In a copy-form value the letters of $w$ occur exactly in the instances;
% a letter $x_j$ occurs once per instance, at offset $j$, and nowhere
% else (constant runs are $\Gamma$-words; an instance spells $w$, whose
% letters are pairwise distinct).  In particular $x_0$ occurs exactly at
% instance starts, and any $n$ consecutive letters spelling $w$ are
% exactly one instance.
%
% If $\pi = 0$, $X$ is a $\Gamma$-word, and a window spelling it cannot
% meet an instance (a non-$\Gamma$ letter would have to equal a
% $\Gamma$-letter); it lies in a single constant run.  If $\pi \geq 1$,
% let the window start at $s$.  $X$'s first non-$\Gamma$ letter is
% $x_0$, at $X$-offset $|u_0|$, so $T[s + |u_0|] = x_0$ is an instance
% start $p_a$; $X$'s first block is $n$ contiguous letters spelling $w$,
% so the stretch $[p_a, p_a + n)$ is exactly instance $a$ and coincides
% with $X$'s first block.  Inductively, if $X$'s $i$-th block coincides
% with instance $a+i-1$, the next $|u_i|$ window letters must spell the
% $\Gamma$-word $u_i$: they cannot be shorter than $v_{a+i-1}$ (the
% following letter of $X$ is $x_0$, non-$\Gamma$, but a shorter gap
% lands inside the $\Gamma$-run $v_{a+i-1}$), nor longer (a longer gap
% reaches into instance $a+i$, whose letters are non-$\Gamma$, or past
% the text's end).  So $u_i = v_{a+i-1}$ and $X$'s $(i{+}1)$-st block
% starts at instance $a+i$'s start, coinciding with it.  The trailing
% gap $u_\pi$ is a prefix of $v_{a+\pi-1}$, the leading gap $u_0$ a
% suffix of $v_{a-1}$.
% \end{proof}
%
% \begin{lemma}[Preservation]
% \label{lem:preserve}
% With $T$, $X$ as in Lemma~\ref{lem:copyalign} and
% $y = z_0 W z_1 \cdots W z_\nu$ in copy form, the value $[y/X]T$ (the
% greedy leftmost-first pass, never rescanning inserted text) is in copy
% form: its instances are the instances of $T$ not covered by any match,
% whole and in order, plus $\nu$ fresh copies of $y$'s instances per
% match site.
% \end{lemma}
%
% \begin{proof}
% The scan selects pairwise disjoint windows, leftmost first, resuming
% after each; inserted text is never re-entered.  By
% Lemma~\ref{lem:copyalign} every selected window is a union of whole
% instances plus pieces of constant runs, so each instance of $T$ is
% covered by exactly one match or by none.  The output interleaves the
% uncovered instances with the surviving constant pieces and, at each
% site, one copy of $y$ -- a copy-form value by hypothesis.  The
% $\Gamma$-pieces and the labeled blocks assemble into copy form.
% \end{proof}
%
% \begin{theorem}[Separable rigidity]
% \label{thm:separable}
% For every expression $E$ (any $S$-depth, any shape) and every separable
% input $w$ with $|w| = n \geq 1$: whenever $\llbracket E \rrbracket(w)$
% is defined it is in copy form, $\llbracket E \rrbracket(w) = v_0 W v_1
% \cdots W v_M$, and
% \[ \mathrm{prov}(E,w) \;=\; (0,1,\dots,n{-}1)^{M}. \]
% \end{theorem}
%
% \begin{proof}
% Induction on the number of nodes of $E$.  Constants are copy form
% with $M = 0$; the variable is $W$ ($M = 1$); concatenation merges two
% copy forms at the junction.  For the pass node, the three children are
% smaller expressions, so $T$, $X$ and $y$ are in copy form by the
% induction hypothesis; if $X = \epsilon$ the node is undefined, and
% otherwise Lemma~\ref{lem:preserve} applies.  In copy form the labeled
% atoms are exactly the instances', each block contributing
% $(0,\dots,n{-}1)$ left to right.
% \end{proof}
%
% \begin{corollary}[No descending provenance, any depth]
% \label{cor:nodb}
% For $n \geq 2$ and $w$ separable, $\mathrm{prov}(E,w)$ is either empty
% or contains the adjacent ascending pair $(0,1)$.  So $E$ realizes no
% descending bijection, and no strictly decreasing provenance of length
% $\geq 2$, at any $S$-depth.  In the budget program's terms: the budget
% conjecture holds on separable inputs with budget $1$ for every $E$.
% \end{corollary}
%
% \begin{corollary}[Reversal fails on every separable input]
% \label{cor:norev}
% For $n \geq 2$ and $w$ separable, $E(w) \neq \mathrm{rev}(w)$: the
% output is either a $\Gamma$-word (while $\mathrm{rev}(w)$ contains
% every letter of $w$) or contains $w$ \emph{forwards} as a contiguous
% block, which $\mathrm{rev}(w)$ -- of the same length -- contains only
% if $w = \mathrm{rev}(w)$, impossible for pairwise distinct letters at
% $n \geq 2$.
% \end{corollary}
%
% \begin{theorem}[Reversal over unbounded alphabets]
% \label{thm:rev-unbounded}
% No expression computes reversal on all strings: for every $E$ there
% are separable inputs of every length $\geq 2$, and $E$ differs from
% reversal on every one of them (Corollary~\ref{cor:norev}); a fortiori
% no $E$ computes reversal on $\Sigma^*$ for every finite $\Sigma$.
% Via the forcing lemma (rounds 10-12): a reversal-computing $E$ must
% realize the descending bijection on every separable input, and
% Corollary~\ref{cor:nodb} excludes that at every depth.  Over a fixed
% alphabet the theorem leaves exactly the constraint
% $|\Sigma \setminus \Gamma(E)| \leq 1$ on any witness -- a witness's
% constants must cover all but one letter of $\Sigma$ -- which by the
% laundering theorem of round 12 is a constraint, not an obstruction;
% the fixed-alphabet question is untouched and remains open.
% \end{theorem}
%
% \emph{Verified} (all runs < 60 s, scripts in
% \texttt{research/scratch/rev-db/}): copy-alignment and instance
% integrity checked at the atom level on 2{,}500 instrumented random
% evaluations (1{,}170 match windows, 0 instance cuts, 0 malformed
% instances, exact agreement with the independent evaluator on every
% case); the prov and content forms on 9{,}300 random expressions to
% $S$-depth 8, over $\Gamma \in \{a,b\}$ and $\{a,b,z\}$
% (0 violations); exhaustive-in-shape falsification at
% $S$-depth $\leq 4$ over a 24-form pass-free library on canonical
% separable inputs -- exhaustive per length up to letter renaming --
% covering chains, deep patterns, deep replacements and
% $C$-compositions: 122{,}935{,}396 defined pipelines, 0 violations of
% copy-alignment, prov form, content form, DB, FDI, or reversal; 1{,}874
% CROSS samples of the sweep re-evaluated with the independent
% evaluator, exact agreement throughout.
%
% Relation to the depth-2 theorems.  Rounds 10-11 bound the DB-realizing
% inputs of depth-<=2 expressions for \emph{all} inputs -- stronger in
% the input dimension, weaker in depth -- and their machinery (picks,
% sandwiches, frames, B-rigidity, residues) is what that stronger
% statement needs.  Separable rigidity is all-depths but
% separable-only; for the reversal application, which by the forcing
% lemma constrains only separable inputs, it suffices alone.  Every DB
% realization on record sits on a non-separable input (repeated letters
% or letters shared with the expression's constants), consistent with
% both.  The budget conjecture for arbitrary inputs at depth $\geq 3$
% remains open.
